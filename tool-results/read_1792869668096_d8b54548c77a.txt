    54→/**
    55→ * EditorPage — the full application shell:
    56→ * header (title + tools) · scrollable document surface · status footer.
    57→ */
    58→
    59→type UploadKind = "image" | "video" | "file";
    60→
    61→export default function EditorPage() {
    62→  const [ready, setReady] = React.useState(false);
    63→  const [title, setTitle] = React.useState("");
    64→  const [searchOpen, setSearchOpen] = React.useState(false);
    65→  const [shortcutsOpen, setShortcutsOpen] = React.useState(false);
    66→  const [emojiAnchor, setEmojiAnchor] = React.useState<DOMRect | null>(null);
    67→  const [embedOpen, setEmbedOpen] = React.useState(false);
    68→  const [contextMenu, setContextMenu] = React.useState<ContextMenuState | null>(null);
    69→
    70→  const contentRef = React.useRef<HTMLDivElement>(null);
    71→  const fileInputRef = React.useRef<HTMLInputElement>(null);
    72→  const uploadKindRef = React.useRef<UploadKind>("image");
    73→  const dropInsertPos = React.useRef<number | undefined>(undefined);
    74→
    75→  /* ------------------------------------------------ helpers (slash actions) */
    76→  const requestUpload = React.useCallback((kind: UploadKind) => {
    77→    uploadKindRef.current = kind;
    78→    const input = fileInputRef.current;
    79→    if (input) {
    80→      input.accept =
    81→        kind === "image" ? "image/*" : kind === "video" ? "video/*" : "*/*";
    82→      input.click();
    83→    }
    84→  }, []);
    85→
    86→  const insertUpload = React.useCallback(
    87→    (
    88→      editor: Editor,
    89→      result: { url: string; name: string; size: number; type: string },
    90→      kind: UploadKind,
    91→      replacePos?: number,
    92→    ) => {
    93→      const node: JSONContent =
    94→        kind === "image"
    95→          ? { type: "imageBlock", attrs: { src: result.url, alt: result.name, width: 100, align: "center" } }
    96→          : kind === "video"
    97→            ? { type: "videoBlock", attrs: { src: result.url, name: result.name } }
    98→            : { type: "fileBlock", attrs: { src: result.url, name: result.name, size: result.size, type: result.type } };
    99→      // A selected media placeholder is REPLACED by the finished block.
   100→      if (replacePos !== undefined) {
   101→        const target = editor.state.doc.nodeAt(replacePos);
   102→        const to = target ? replacePos + target.nodeSize : replacePos + 1;
   103→        editor.chain().focus().insertContentAt({ from: replacePos, to }, node).run();
   104→        return;
   105→      }
   106→      editor
   107→        .chain()
   108→        .focus()
   109→        .insertContentAt(dropInsertPos.current ?? editor.state.selection.to, node)
   110→        .run();
   111→      dropInsertPos.current = undefined;
   112→    },
   113→    [],
   114→  );
   115→
   116→  const handleFiles = React.useCallback(
   117→    async (files: File[], editorInstance: Editor, insertPos?: number, replacePos?: number) => {
   118→      dropInsertPos.current = insertPos;
   119→      for (const file of files) {
   120→        const loadingToast = toast.loading(`Uploading ${file.name}…`, {
   121→          description: formatBytes(file.size),
   122→        });
   123→        try {
   124→          const result = await uploadFile(file);
   125→          toast.dismiss(loadingToast);
   126→          // Choose the block type from the actual mime type (handles
   127→          // drag-drop where the slash-menu kind is not authoritative).
   128→          const kind: UploadKind = file.type.startsWith("image")
   129→            ? "image"
   130→            : file.type.startsWith("video")
   131→              ? "video"
   132→              : "file";
   133→          insertUpload(editorInstance, result, kind, replacePos);
   134→          toast.success(`${file.name} added`);
   135→        } catch (error) {
   136→          toast.dismiss(loadingToast);
   137→          toast.error(`Upload failed: ${error instanceof Error ? error.message : "unknown error"}`);
   138→        }
   139→      }
   140→    },
   141→    [insertUpload],
   142→  );
   143→
   144→  /** If the current selection is a media placeholder, its document position. */
   145→  const selectedPlaceholderPos = React.useCallback((editorInstance: Editor): number | undefined => {
   146→    const { selection } = editorInstance.state;
   147→    if (selection instanceof NodeSelection && selection.node.type.name === "mediaPlaceholder") {
   148→      return selection.from;
   149→    }
   150→    return undefined;
   151→  }, []);
   152→
   153→  /** Commit a URL into the selected media placeholder (paste-a-link UX). */
   154→  const commitPlaceholderUrl = React.useCallback(
   155→    (editorInstance: Editor, pos: number, rawUrl: string): boolean => {
   156→      const node = editorInstance.state.doc.nodeAt(pos);
   157→      if (!node || node.type.name !== "mediaPlaceholder") return false;
   158→      const rawKind = node.attrs.kind;
   159→      const kind: MediaKind = isMediaKind(rawKind) ? rawKind : "image";
   160→      const url = /^https?:\/\//i.test(rawUrl.trim()) ? rawUrl.trim() : `https://${rawUrl.trim()}`;
   161→      let name = "";
   162→      try {
   163→        const parsed = new URL(url);
   164→        name =
   165→          decodeURIComponent(parsed.pathname.split("/").filter(Boolean).pop() ?? "") || parsed.hostname;
   166→      } catch {
   167→        return false;
   168→      }
   169→      editorInstance
   170→        .chain()
   171→        .focus()
   172→        .insertContentAt(
   173→          { from: pos, to: pos + node.nodeSize },
   174→          mediaNodeFor(kind, {
   175→            url,
   176→            name,
   177→            size: 0,
   178→            type: kind === "image" ? "image/remote" : kind === "video" ? "video/remote" : guessTypeFromUrl(url),
   179→          }),
   180→        )
   181→        .run();
   182→      return true;
   183→    },
   184→    [],
   185→  );
   186→
   187→  /* ------------------------------------------------ slash action helpers */
   188→  const requestEmoji = React.useCallback((editorInstance: Editor) => {
   189→    const coords = editorInstance.view.coordsAtPos(editorInstance.state.selection.to);
   190→    setEmojiAnchor(new DOMRect(coords.left, coords.top, 0, 0));
   191→  }, []);
   192→
   193→  const requestEmbed = React.useCallback(() => setEmbedOpen(true), []);
   194→
   195→  const requestMath = React.useCallback((editorInstance: Editor, inline: boolean) => {
   196→    if (inline) editorInstance.chain().focus().insertMathInline("").run();
   197→    else editorInstance.chain().focus().insertMathBlock("").run();
   198→  }, []);
   199→
   200→  // Stable across renders — safe to capture once inside useEditor options.
   201→  const slashHelpers = React.useMemo<SlashActionHelpers>(
   202→    () => ({ requestUpload, requestEmoji, requestEmbed, requestMath }),
   203→    [requestUpload, requestEmoji, requestEmbed, requestMath],
   204→  );
   205→
   206→  /* --------------------------------------------- mention overlays (page/person) */
   207→  const [openPageId, setOpenPageId] = React.useState<string | null>(null);
   208→  const [profilePerson, setProfilePerson] = React.useState<{ id: string; label?: string } | null>(null);
   209→
   210→  React.useEffect(() => {
   211→    const onOpenPage = (e: Event) => setOpenPageId((e as CustomEvent<string>).detail);
   212→    const onShowPerson = (e: Event) => {
   213→      const id = (e as CustomEvent<string>).detail;
   214→      setProfilePerson({ id });
   215→    };
   216→    window.addEventListener("aurora:open-page", onOpenPage);
   217→    window.addEventListener("aurora:show-person", onShowPerson);
   218→    return () => {
   219→      window.removeEventListener("aurora:open-page", onOpenPage);
   220→      window.removeEventListener("aurora:show-person", onShowPerson);
   221→    };
   222→  }, []);
   223→
   224→  /* Deep link — /?page=<id> opens the page viewer (new-tab mentions). */
   225→  React.useEffect(() => {
   226→    const params = new URLSearchParams(window.location.search);
   227→    const pageId = params.get("page");
   228→    if (pageId) setOpenPageId(pageId);
   229→  }, []);
   230→
   231→  // The editor instance lands in a ref *after* creation so paste/drop
   232→  // handlers (created before the editor exists) can reach it safely.
   233→  const editorRef = React.useRef<Editor | null>(null);
   234→
   235→  /* ------------------------------------------------ editor instance */
   236→  const editor = useEditor(
   237→    {
   238→      extensions: buildExtensions({
   239→        getActionContext: () => slashHelpers,
   240→      }),
   241→      content: { type: "doc", content: [{ type: "paragraph" }] },
   242→      immediatelyRender: false,
   243→      autofocus: false,
   244→      editorProps: {
   245→        attributes: {
   246→          class: "tiptap",
   247→          "aria-label": "Document editor",
   248→          role: "textbox",
   249→        },
   250→        handlePaste: (_view, event) => {
   251→          const editorInstance = editorRef.current;
   252→          if (!editorInstance) return false;
   253→          const clipboard = event.clipboardData;
   254→          const files = Array.from(clipboard?.files ?? []);
   255→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   256→
   257→          // Pasting a URL onto a selected placeholder commits it directly.
   258→          if (placeholderPos !== undefined && !files.length) {
   259→            const text = (clipboard?.getData("text/plain") ?? "").trim();
   260→            if (/^(https?:\/\/|www\.)\S+$/i.test(text) && commitPlaceholderUrl(editorInstance, placeholderPos, text)) {
   261→              return true;
   262→            }
   263→          }
   264→
   265→          if (!files.length) return false;
   266→          // Screenshots & copied files — pick the block kind by mime type.
   267→          for (const file of files) {
   268→            void handleFiles([file], editorInstance, undefined, placeholderPos);
   269→          }
   270→          return true;
   271→        },
   272→        handleDrop: (view, event, _slice, moved) => {
   273→          const editorInstance = editorRef.current;
   274→          if (moved || !editorInstance) return false;
   275→          // Drops on a media placeholder card are handled by the card itself.
   276→          if ((event.target as HTMLElement)?.closest?.("[data-media-placeholder-ui]")) return false;
   277→          const files = Array.from(event.dataTransfer?.files ?? []);
   278→          if (!files.length) return false;
   279→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   280→          const coords = view.posAtCoords({ left: event.clientX, top: event.clientY });
   281→          for (const file of files) {
   282→            void handleFiles([file], editorInstance, coords?.pos, placeholderPos);
   283→          }
   284→          return true;
   285→        },
   286→      },
   287→    },
   288→    [],
   289→  );
   290→
   291→  /* ------------------------------------------------ keep editorRef fresh */
   292→  React.useEffect(() => {
   293→    editorRef.current = editor;
   294→    // Debug handle for browser console testing (dev only).
   295→    if (process.env.NODE_ENV !== "production") {
   296→      (window as unknown as { __editor?: Editor }).__editor = editor;
   297→    }
   298→  }, [editor]);
   299→
   300→  /* ------------------------------------------------ load document */
   301→  React.useEffect(() => {
   302→    if (!editor) return;
   303→    let cancelled = false;
   304→    fetchDocument()
   305→      .then((doc) => {
   306→        if (cancelled) return;
   307→        setTitle(doc.title);
   308→        if (doc.content && typeof doc.content === "object" && "type" in (doc.content as object)) {
   309→          editor.commands.setContent(doc.content as JSONContent, { emitUpdate: false });
   310→        }
   311→        setReady(true);
   312→      })
   313→      .catch((error) => {
   314→        if (cancelled) return;
   315→        console.error("[editor] failed to load document", error);
   316→        toast.error("Couldn't load your document", {
   317→          description: "You can keep writing — changes will retry on save.",
   318→        });
   319→        setReady(true);
   320→      });
   321→    return () => {
   322→      cancelled = true;
   323→    };
   324→  }, [editor]);
   325→
   326→  /* ------------------------------------------------ autosave */
   327→  const { status, savedAt, saveNow } = useAutosave({ editor, title, ready });
   328→
   329→  /* ------------------------------------------------ global shortcuts */
   330→  React.useEffect(() => {
   331→    const onKey = (e: KeyboardEvent) => {
   332→      const mod = e.metaKey || e.ctrlKey;
   333→      if (mod && e.key.toLowerCase() === "f") {
   334→        e.preventDefault();
   335→        setSearchOpen(true);
   336→      }
   337→      if (mod && e.key === "/") {
   338→        e.preventDefault();
   339→        setShortcutsOpen((v) => !v);
   340→      }
   341→    };
   342→    window.addEventListener("keydown", onKey);
   343→    return () => window.removeEventListener("keydown", onKey);
   344→  }, []);
   345→
   346→  /* ------------------------------------------------ future-ready stubs */
   347→  React.useEffect(() => {
   348→    const onFutureFeature = (e: Event) => {
   349→      const detail = (e as CustomEvent<string>).detail ?? "This feature";
   350→      toast(`${detail} — coming soon`, {
   351→        description: "The extension registry is ready for it.",
   352→      });
   353→    };
   354→    window.addEventListener("aurora:future-feature", onFutureFeature);
   355→    return () => window.removeEventListener("aurora:future-feature", onFutureFeature);
   356→  }, []);
   357→
   358→  /* ------------------------------------------------ demo & misc actions */
   359→  const loadDemo = React.useCallback(() => {
   360→    if (!editor) return;
   361→    if (editor.isEmpty) {
   362→      editor.commands.setContent(DEMO_CONTENT, { emitUpdate: true });
   363→    } else {
   364→      editor
   365→        .chain()
   366→        .focus("end")
   367→        .insertContentAt(editor.state.doc.content.size, DEMO_CONTENT.content)
   368→        .run();
   369→    }
   370→    toast.success("Demo document loaded", {
   371→      description: "Hover blocks, press / anywhere, select text — explore!",
   372→    });
   373→  }, [editor]);
   374→
   375→  return (
   376→    <div className="flex h-screen flex-col overflow-hidden bg-background">
   377→      <Header
   378→        editor={editor}
   379→        title={title}
   380→        onTitleChange={setTitle}
   381→        status={status}
   382→        onUndo={() => editor?.chain().focus().undo().run()}
   383→        onRedo={() => editor?.chain().focus().redo().run()}
   384→        onOpenSearch={() => setSearchOpen(true)}
   385→        onOpenShortcuts={() => setShortcutsOpen(true)}
   386→        onLoadDemo={loadDemo}
   387→      />
   388→
   389→      <main className="relative flex-1 overflow-y-auto" aria-label="Document">
   390→        {/* Loading skeleton */}
   391→        {!ready && (
   392→          <div className="pointer-events-none absolute inset-x-0 top-10 z-10 mx-auto max-w-[720px] px-6">
   393→            <EditorSkeleton />
   394→          </div>
   395→        )}
   396→
   397→        <div ref={contentRef} className="relative mx-auto w-full max-w-[880px] px-5 pb-24 pt-12 md:px-20 lg:px-24">
   398→          <div
   399→            data-editor-column
   400→            className="rich-editor mx-auto max-w-[720px]"
   401→            onContextMenu={(e) => {
   402→              if (!editor) return;
   403→              const coords = editor.view.posAtCoords({ left: e.clientX, top: e.clientY });
   404→              if (!coords) return;
   405→              e.preventDefault();
   406→              setContextMenu({ x: e.clientX, y: e.clientY, pos: coords.pos });
   407→            }}
   408→          >
   409→            {editor && (
   410→              <>
   411→                <EditorContent editor={editor} />
   412→                <FloatingToolbar editor={editor} />
   413→                <TableMenu editor={editor} />
   414→              </>
   415→            )}
   416→          </div>
   417→
   418→          {editor && ready && <BlockHandle editor={editor} contentRef={contentRef} />}
   419→          {editor && ready && <EmptyStateHint editor={editor} onLoadDemo={loadDemo} />}
   420→        </div>
   421→      </main>
   422→
   423→      <StatusBar editor={editor} status={status} savedAt={savedAt} />
   424→
   425→      {/* ------------------------------ overlays ------------------------------ */}
   426→      {editor && <SlashMenu />}
   427→      {editor && <AtMenu editor={editor} />}
   428→      {editor && <EmojiMenu />}
   429→      {editor && <SearchPanel editor={editor} open={searchOpen} onClose={() => setSearchOpen(false)} />}
   430→      {editor && (
   431→        <ContextMenu
   432→          editor={editor}
   433→          state={contextMenu}
   434→          onClose={() => setContextMenu(null)}
   435→        />
   436→      )}
   437→      {editor && <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />}
   438→
   439→      {editor && emojiAnchor && (
   440→        <EmojiPicker
   441→          anchorRect={emojiAnchor}
   442→          onPick={(emoji) => {
   443→            editor.chain().focus().insertContent(emoji).run();
   444→            setEmojiAnchor(null);
   445→          }}
   446→          onClose={() => setEmojiAnchor(null)}
   447→        />
   448→      )}
   449→
   450→      <EmbedDialog
   451→        open={embedOpen}
   452→        onOpenChange={setEmbedOpen}
   453→        onSubmit={(url) => {
   454→          const normalized = normalizeEmbedUrl(url);
   455→          if (!normalized) {
   456→            toast.error("That doesn't look like a valid embed URL");
   457→            return;
   458→          }
   459→          editor?.chain().focus().insertEmbed(normalized).run();
   460→          setEmbedOpen(false);
   461→          toast.success("Embed added");
   462→        }}
   463→      />
   464→
   465→      {/* Mention destinations */}
   466→      {openPageId ? <PageViewer pageId={openPageId} onClose={() => setOpenPageId(null)} /> : null}
   467→      {profilePerson ? (
   468→        <PersonProfileDialog
   469→          personId={profilePerson.id}
   470→          fallbackName={profilePerson.label}
   471→          onClose={() => setProfilePerson(null)}
   472→        />
   473→      ) : null}
   474→
   475→      {/* Hidden file input for uploads */}
   476→      <input
   477→        ref={fileInputRef}
   478→        type="file"
   479→        multiple
   480→        className="hidden"
   481→        onChange={(e) => {
   482→          const files = Array.from(e.target.files ?? []);
   483→          if (files.length && editor) void handleFiles(files, editor);
   484→          e.target.value = "";
   485→        }}
   486→      />
   487→    </div>
   488→  );
   489→}

   491→/* ================================================================ Header */
   492→
   493→function Header({
   494→  editor,
   495→  title,
   496→  onTitleChange,
   497→  status,
   498→  onUndo,
   499→  onRedo,
   500→  onOpenSearch,
   501→  onOpenShortcuts,
   502→  onLoadDemo,
   503→}: {
   504→  editor: Editor | null;
   505→  title: string;
   506→  onTitleChange: (value: string) => void;
   507→  status: SaveStatus;
   508→  onUndo: () => void;
   509→  onRedo: () => void;
   510→  onOpenSearch: () => void;
   511→  onOpenShortcuts: () => void;
   512→  onLoadDemo: () => void;
   513→}) {
   514→  const { theme, setTheme } = useTheme();
   515→  const [mounted, setMounted] = React.useState(false);
   516→  React.useEffect(() => setMounted(true), []);
   517→
   518→  const history = useEditorState({
   519→    editor,
   520→    selector: ({ editor: e }) =>
   521→      e ? { canUndo: e.can().undo(), canRedo: e.can().redo() } : { canUndo: false, canRedo: false },
   522→  });
   523→
   524→  return (
   525→    <TooltipProvider delayDuration={300}>
   526→      <header className="sticky top-0 z-50 shrink-0 border-b border-border/60 bg-background/85 backdrop-blur-xl">
   527→        <div className="mx-auto flex h-[52px] w-full max-w-5xl items-center gap-2 px-3 md:px-5">
   528→          <div
   529→            className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500 via-purple-500 to-fuchsia-500 text-white shadow-md shadow-violet-500/25"
   530→            aria-hidden
   531→          >
   532→            <Feather size={15} strokeWidth={2.2} />
   533→          </div>
   534→
   535→          <input
   536→            value={title}
   537→            onChange={(e) => onTitleChange(e.target.value)}
   538→            onKeyDown={(e) => {
   539→              if (e.key === "Enter") {
   540→                e.preventDefault();
   541→                editor?.commands.focus("start");
   542→              }
   543→            }}
   544→            placeholder="Untitled"
   545→            aria-label="Document title"
   546→            className="min-w-0 flex-1 rounded-md bg-transparent px-1.5 py-1 text-[14.5px] font-semibold tracking-tight outline-none transition-colors placeholder:text-muted-foreground/50 hover:bg-muted/40 focus:bg-muted/60"
   547→          />
   548→
   549→          <SaveChip status={status} />
   550→
   551→          <div className="flex items-center gap-0.5">
   552→            <HeaderBtn title="Undo — ⌘Z" onClick={onUndo} disabled={!history?.canUndo}>
   553→              <Undo2 size={15} />
   554→            </HeaderBtn>
   555→            <HeaderBtn title="Redo — ⌘⇧Z" onClick={onRedo} disabled={!history?.canRedo}>
   556→              <Redo2 size={15} />
   557→            </HeaderBtn>
   558→
   559→            <span className="mx-1 h-5 w-px bg-border/70" aria-hidden />
   560→

   640→function StatusBar({
   641→  editor,
   642→  status,
   643→  savedAt,
   644→}: {
   645→  editor: Editor | null;
   646→  status: SaveStatus;
   647→  savedAt: Date | null;
   648→}) {
   649→  const stats = useEditorState({
   650→    editor,
   651→    selector: ({ editor: e }) =>
   652→      e
   653→        ? {
   654→            words: e.storage.characterCount.words(),
   655→            chars: e.storage.characterCount.characters(),
   656→            blocks: e.state.doc.childCount,
   657→          }
   658→        : { words: 0, chars: 0, blocks: 0 },
   659→  });
   660→
   661→  const timeLabel = savedAt
   662→    ? `· ${relativeTime(savedAt)}`
   663→    : "";
   664→
   665→  return (
   666→    <footer className="mt-auto shrink-0 border-t border-border/60 bg-background/85 backdrop-blur-xl">
   667→      <div className="mx-auto flex h-9 w-full max-w-5xl items-center justify-between px-4 pb-[env(safe-area-inset-bottom)] text-[11.5px] text-muted-foreground">
   668→        <div className="flex items-center gap-3 tabular-nums">
   669→          <span>
   670→            <span className="font-medium text-foreground/70">{stats?.blocks ?? 0}</span> blocks
   671→          </span>
   672→          <span aria-hidden>·</span>
   673→          <span>
   674→            <span className="font-medium text-foreground/70">{stats?.words ?? 0}</span> words
   675→          </span>
   676→          <span aria-hidden className="hidden sm:inline">·</span>
   677→          <span className="hidden sm:inline">
   678→            <span className="font-medium text-foreground/70">{stats?.chars ?? 0}</span> characters
   679→          </span>
   680→        </div>
   681→        <div className="flex items-center gap-2">
   682→          <span className="hidden md:inline">Press / for commands · ⌘/ shortcuts</span>
   683→          <span aria-hidden className="hidden md:inline">·</span>
   684→          <span className="flex items-center gap-1.5" role="status" aria-live="polite">
   685→            <span
   686→              className={cn(
   687→                "h-1.5 w-1.5 rounded-full",
   688→                status === "saved" && "bg-emerald-500",
   689→                status === "saving" && "animate-pulse bg-violet-400",
   690→                status === "dirty" && "bg-amber-500",
   691→                status === "error" && "bg-red-400",
   692→                (status === "idle") && "bg-muted-foreground/50",
   693→              )}
   694→            />
   695→            {status === "saved" ? `Saved ${timeLabel}` : status === "idle" ? "All changes captured" : status.charAt(0).toUpperCase() + status.slice(1)}
   696→          </span>
   697→        </div>
   698→      </div>
   699→    </footer>
   700→  );
   701→}

   703→function relativeTime(date: Date): string {
   704→  const seconds = Math.round((Date.now() - date.getTime()) / 1000);
   705→  if (seconds < 10) return "just now";
   706→  if (seconds < 60) return `${seconds}s ago`;
   707→  const minutes = Math.round(seconds / 60);
   708→  if (minutes < 60) return `${minutes}m ago`;
   709→  const hours = Math.round(minutes / 60);
   710→  if (hours < 24) return `${hours}h ago`;
   711→  return date.toLocaleDateString();
   712→}
