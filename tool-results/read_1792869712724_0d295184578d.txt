    54→    54→/**
    55→    55→ * EditorPage — the full application shell:
    56→    56→ * header (title + tools) · scrollable document surface · status footer.
    57→    57→ */
    58→    58→
    59→    59→type UploadKind = "image" | "video" | "file";
    60→    60→
    61→    61→export default function EditorPage() {
    62→    62→  const [ready, setReady] = React.useState(false);
    63→    63→  const [title, setTitle] = React.useState("");
    64→    64→  const [searchOpen, setSearchOpen] = React.useState(false);
    65→    65→  const [shortcutsOpen, setShortcutsOpen] = React.useState(false);
    66→    66→  const [emojiAnchor, setEmojiAnchor] = React.useState<DOMRect | null>(null);
    67→    67→  const [embedOpen, setEmbedOpen] = React.useState(false);
    68→    68→  const [contextMenu, setContextMenu] = React.useState<ContextMenuState | null>(null);
    69→    69→
    70→    70→  const contentRef = React.useRef<HTMLDivElement>(null);
    71→    71→  const fileInputRef = React.useRef<HTMLInputElement>(null);
    72→    72→  const uploadKindRef = React.useRef<UploadKind>("image");
    73→    73→  const dropInsertPos = React.useRef<number | undefined>(undefined);
    74→    74→
    75→    75→  /* ------------------------------------------------ helpers (slash actions) */
    76→    76→  const requestUpload = React.useCallback((kind: UploadKind) => {
    77→    77→    uploadKindRef.current = kind;
    78→    78→    const input = fileInputRef.current;
    79→    79→    if (input) {
    80→    80→      input.accept =
    81→    81→        kind === "image" ? "image/*" : kind === "video" ? "video/*" : "*/*";
    82→    82→      input.click();
    83→    83→    }
    84→    84→  }, []);
    85→    85→
    86→    86→  const insertUpload = React.useCallback(
    87→    87→    (
    88→    88→      editor: Editor,
    89→    89→      result: { url: string; name: string; size: number; type: string },
    90→    90→      kind: UploadKind,
    91→    91→      replacePos?: number,
    92→    92→    ) => {
    93→    93→      const node: JSONContent =
    94→    94→        kind === "image"
    95→    95→          ? { type: "imageBlock", attrs: { src: result.url, alt: result.name, width: 100, align: "center" } }
    96→    96→          : kind === "video"
    97→    97→            ? { type: "videoBlock", attrs: { src: result.url, name: result.name } }
    98→    98→            : { type: "fileBlock", attrs: { src: result.url, name: result.name, size: result.size, type: result.type } };
    99→    99→      // A selected media placeholder is REPLACED by the finished block.
   100→   100→      if (replacePos !== undefined) {
   101→   101→        const target = editor.state.doc.nodeAt(replacePos);
   102→   102→        const to = target ? replacePos + target.nodeSize : replacePos + 1;
   103→   103→        editor.chain().focus().insertContentAt({ from: replacePos, to }, node).run();
   104→   104→        return;
   105→   105→      }
   106→   106→      editor
   107→   107→        .chain()
   108→   108→        .focus()
   109→   109→        .insertContentAt(dropInsertPos.current ?? editor.state.selection.to, node)
   110→   110→        .run();
   111→   111→      dropInsertPos.current = undefined;
   112→   112→    },
   113→   113→    [],
   114→   114→  );
   115→   115→
   116→   116→  const handleFiles = React.useCallback(
   117→   117→    async (files: File[], editorInstance: Editor, insertPos?: number, replacePos?: number) => {
   118→   118→      dropInsertPos.current = insertPos;
   119→   119→      for (const file of files) {
   120→   120→        const loadingToast = toast.loading(`Uploading ${file.name}…`, {
   121→   121→          description: formatBytes(file.size),
   122→   122→        });
   123→   123→        try {
   124→   124→          const result = await uploadFile(file);
   125→   125→          toast.dismiss(loadingToast);
   126→   126→          // Choose the block type from the actual mime type (handles
   127→   127→          // drag-drop where the slash-menu kind is not authoritative).
   128→   128→          const kind: UploadKind = file.type.startsWith("image")
   129→   129→            ? "image"
   130→   130→            : file.type.startsWith("video")
   131→   131→              ? "video"
   132→   132→              : "file";
   133→   133→          insertUpload(editorInstance, result, kind, replacePos);
   134→   134→          toast.success(`${file.name} added`);
   135→   135→        } catch (error) {
   136→   136→          toast.dismiss(loadingToast);
   137→   137→          toast.error(`Upload failed: ${error instanceof Error ? error.message : "unknown error"}`);
   138→   138→        }
   139→   139→      }
   140→   140→    },
   141→   141→    [insertUpload],
   142→   142→  );
   143→   143→
   144→   144→  /** If the current selection is a media placeholder, its document position. */
   145→   145→  const selectedPlaceholderPos = React.useCallback((editorInstance: Editor): number | undefined => {
   146→   146→    const { selection } = editorInstance.state;
   147→   147→    if (selection instanceof NodeSelection && selection.node.type.name === "mediaPlaceholder") {
   148→   148→      return selection.from;
   149→   149→    }
   150→   150→    return undefined;
   151→   151→  }, []);
   152→   152→
   153→   153→  /** Commit a URL into the selected media placeholder (paste-a-link UX). */
   154→   154→  const commitPlaceholderUrl = React.useCallback(
   155→   155→    (editorInstance: Editor, pos: number, rawUrl: string): boolean => {
   156→   156→      const node = editorInstance.state.doc.nodeAt(pos);
   157→   157→      if (!node || node.type.name !== "mediaPlaceholder") return false;
   158→   158→      const rawKind = node.attrs.kind;
   159→   159→      const kind: MediaKind = isMediaKind(rawKind) ? rawKind : "image";
   160→   160→      const url = /^https?:\/\//i.test(rawUrl.trim()) ? rawUrl.trim() : `https://${rawUrl.trim()}`;
   161→   161→      let name = "";
   162→   162→      try {
   163→   163→        const parsed = new URL(url);
   164→   164→        name =
   165→   165→          decodeURIComponent(parsed.pathname.split("/").filter(Boolean).pop() ?? "") || parsed.hostname;
   166→   166→      } catch {
   167→   167→        return false;
   168→   168→      }
   169→   169→      editorInstance
   170→   170→        .chain()
   171→   171→        .focus()
   172→   172→        .insertContentAt(
   173→   173→          { from: pos, to: pos + node.nodeSize },
   174→   174→          mediaNodeFor(kind, {
   175→   175→            url,
   176→   176→            name,
   177→   177→            size: 0,
   178→   178→            type: kind === "image" ? "image/remote" : kind === "video" ? "video/remote" : guessTypeFromUrl(url),
   179→   179→          }),
   180→   180→        )
   181→   181→        .run();
   182→   182→      return true;
   183→   183→    },
   184→   184→    [],
   185→   185→  );
   186→   186→
   187→   187→  /* ------------------------------------------------ slash action helpers */
   188→   188→  const requestEmoji = React.useCallback((editorInstance: Editor) => {
   189→   189→    const coords = editorInstance.view.coordsAtPos(editorInstance.state.selection.to);
   190→   190→    setEmojiAnchor(new DOMRect(coords.left, coords.top, 0, 0));
   191→   191→  }, []);
   192→   192→
   193→   193→  const requestEmbed = React.useCallback(() => setEmbedOpen(true), []);
   194→   194→
   195→   195→  const requestMath = React.useCallback((editorInstance: Editor, inline: boolean) => {
   196→   196→    if (inline) editorInstance.chain().focus().insertMathInline("").run();
   197→   197→    else editorInstance.chain().focus().insertMathBlock("").run();
   198→   198→  }, []);
   199→   199→
   200→   200→  // Stable across renders — safe to capture once inside useEditor options.
   201→   201→  const slashHelpers = React.useMemo<SlashActionHelpers>(
   202→   202→    () => ({ requestUpload, requestEmoji, requestEmbed, requestMath }),
   203→   203→    [requestUpload, requestEmoji, requestEmbed, requestMath],
   204→   204→  );
   205→   205→
   206→   206→  /* --------------------------------------------- mention overlays (page/person) */
   207→   207→  const [openPageId, setOpenPageId] = React.useState<string | null>(null);
   208→   208→  const [profilePerson, setProfilePerson] = React.useState<{ id: string; label?: string } | null>(null);
   209→   209→
   210→   210→  React.useEffect(() => {
   211→   211→    const onOpenPage = (e: Event) => setOpenPageId((e as CustomEvent<string>).detail);
   212→   212→    const onShowPerson = (e: Event) => {
   213→   213→      const id = (e as CustomEvent<string>).detail;
   214→   214→      setProfilePerson({ id });
   215→   215→    };
   216→   216→    window.addEventListener("aurora:open-page", onOpenPage);
   217→   217→    window.addEventListener("aurora:show-person", onShowPerson);
   218→   218→    return () => {
   219→   219→      window.removeEventListener("aurora:open-page", onOpenPage);
   220→   220→      window.removeEventListener("aurora:show-person", onShowPerson);
   221→   221→    };
   222→   222→  }, []);
   223→   223→
   224→   224→  /* Deep link — /?page=<id> opens the page viewer (new-tab mentions). */
   225→   225→  React.useEffect(() => {
   226→   226→    const params = new URLSearchParams(window.location.search);
   227→   227→    const pageId = params.get("page");
   228→   228→    if (pageId) setOpenPageId(pageId);
   229→   229→  }, []);
   230→   230→
   231→   231→  // The editor instance lands in a ref *after* creation so paste/drop
   232→   232→  // handlers (created before the editor exists) can reach it safely.
   233→   233→  const editorRef = React.useRef<Editor | null>(null);
   234→   234→
   235→   235→  /* ------------------------------------------------ editor instance */
   236→   236→  const editor = useEditor(
   237→   237→    {
   238→   238→      extensions: buildExtensions({
   239→   239→        getActionContext: () => slashHelpers,
   240→   240→      }),
   241→   241→      content: { type: "doc", content: [{ type: "paragraph" }] },
   242→   242→      immediatelyRender: false,
   243→   243→      autofocus: false,
   244→   244→      editorProps: {
   245→   245→        attributes: {
   246→   246→          class: "tiptap",
   247→   247→          "aria-label": "Document editor",
   248→   248→          role: "textbox",
   249→   249→        },
   250→   250→        handlePaste: (_view, event) => {
   251→   251→          const editorInstance = editorRef.current;
   252→   252→          if (!editorInstance) return false;
   253→   253→          const clipboard = event.clipboardData;
   254→   254→          const files = Array.from(clipboard?.files ?? []);
   255→   255→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   256→   256→
   257→   257→          // Pasting a URL onto a selected placeholder commits it directly.
   258→   258→          if (placeholderPos !== undefined && !files.length) {
   259→   259→            const text = (clipboard?.getData("text/plain") ?? "").trim();
   260→   260→            if (/^(https?:\/\/|www\.)\S+$/i.test(text) && commitPlaceholderUrl(editorInstance, placeholderPos, text)) {
   261→   261→              return true;
   262→   262→            }
   263→   263→          }
   264→   264→
   265→   265→          if (!files.length) return false;
   266→   266→          // Screenshots & copied files — pick the block kind by mime type.
   267→   267→          for (const file of files) {
   268→   268→            void handleFiles([file], editorInstance, undefined, placeholderPos);
   269→   269→          }
   270→   270→          return true;
   271→   271→        },
   272→   272→        handleDrop: (view, event, _slice, moved) => {
   273→   273→          const editorInstance = editorRef.current;
   274→   274→          if (moved || !editorInstance) return false;
   275→   275→          // Drops on a media placeholder card are handled by the card itself.
   276→   276→          if ((event.target as HTMLElement)?.closest?.("[data-media-placeholder-ui]")) return false;
   277→   277→          const files = Array.from(event.dataTransfer?.files ?? []);
   278→   278→          if (!files.length) return false;
   279→   279→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   280→   280→          const coords = view.posAtCoords({ left: event.clientX, top: event.clientY });
   281→   281→          for (const file of files) {
   282→   282→            void handleFiles([file], editorInstance, coords?.pos, placeholderPos);
   283→   283→          }
   284→   284→          return true;
   285→   285→        },
   286→   286→      },
   287→   287→    },
   288→   288→    [],
   289→   289→  );
   290→   290→
   291→   291→  /* ------------------------------------------------ keep editorRef fresh */
   292→   292→  React.useEffect(() => {
   293→   293→    editorRef.current = editor;
   294→   294→    // Debug handle for browser console testing (dev only).
   295→   295→    if (process.env.NODE_ENV !== "production") {
   296→   296→      (window as unknown as { __editor?: Editor }).__editor = editor;
   297→   297→    }
   298→   298→  }, [editor]);
   299→   299→
   300→   300→  /* ------------------------------------------------ load document */
   301→   301→  React.useEffect(() => {
   302→   302→    if (!editor) return;
   303→   303→    let cancelled = false;
   304→   304→    fetchDocument()
   305→   305→      .then((doc) => {
   306→   306→        if (cancelled) return;
   307→   307→        setTitle(doc.title);
   308→   308→        if (doc.content && typeof doc.content === "object" && "type" in (doc.content as object)) {
   309→   309→          editor.commands.setContent(doc.content as JSONContent, { emitUpdate: false });
   310→   310→        }
   311→   311→        setReady(true);
   312→   312→      })
   313→   313→      .catch((error) => {
   314→   314→        if (cancelled) return;
   315→   315→        console.error("[editor] failed to load document", error);
   316→   316→        toast.error("Couldn't load your document", {
   317→   317→          description: "You can keep writing — changes will retry on save.",
   318→   318→        });
   319→   319→        setReady(true);
   320→   320→      });
   321→   321→    return () => {
   322→   322→      cancelled = true;
   323→   323→    };
   324→   324→  }, [editor]);
   325→   325→
   326→   326→  /* ------------------------------------------------ autosave */
   327→   327→  const { status, savedAt, saveNow } = useAutosave({ editor, title, ready });
   328→   328→
   329→   329→  /* ------------------------------------------------ global shortcuts */
   330→   330→  React.useEffect(() => {
   331→   331→    const onKey = (e: KeyboardEvent) => {
   332→   332→      const mod = e.metaKey || e.ctrlKey;
   333→   333→      if (mod && e.key.toLowerCase() === "f") {
   334→   334→        e.preventDefault();
   335→   335→        setSearchOpen(true);
   336→   336→      }
   337→   337→      if (mod && e.key === "/") {
   338→   338→        e.preventDefault();
   339→   339→        setShortcutsOpen((v) => !v);
   340→   340→      }
   341→   341→    };
   342→   342→    window.addEventListener("keydown", onKey);
   343→   343→    return () => window.removeEventListener("keydown", onKey);
   344→   344→  }, []);
   345→   345→
   346→   346→  /* ------------------------------------------------ future-ready stubs */
   347→   347→  React.useEffect(() => {
   348→   348→    const onFutureFeature = (e: Event) => {
   349→   349→      const detail = (e as CustomEvent<string>).detail ?? "This feature";
   350→   350→      toast(`${detail} — coming soon`, {
   351→   351→        description: "The extension registry is ready for it.",
   352→   352→      });
   353→   353→    };
   354→   354→    window.addEventListener("aurora:future-feature", onFutureFeature);
   355→   355→    return () => window.removeEventListener("aurora:future-feature", onFutureFeature);
   356→   356→  }, []);
   357→   357→
   358→   358→  /* ------------------------------------------------ demo & misc actions */
   359→   359→  const loadDemo = React.useCallback(() => {
   360→   360→    if (!editor) return;
   361→   361→    if (editor.isEmpty) {
   362→   362→      editor.commands.setContent(DEMO_CONTENT, { emitUpdate: true });
   363→   363→    } else {
   364→   364→      editor
   365→   365→        .chain()
   366→   366→        .focus("end")
   367→   367→        .insertContentAt(editor.state.doc.content.size, DEMO_CONTENT.content)
   368→   368→        .run();
   369→   369→    }
   370→   370→    toast.success("Demo document loaded", {
   371→   371→      description: "Hover blocks, press / anywhere, select text — explore!",
   372→   372→    });
   373→   373→  }, [editor]);
   374→   374→
   375→   375→  return (
   376→   376→    <div className="flex h-screen flex-col overflow-hidden bg-background">
   377→   377→      <Header
   378→   378→        editor={editor}
   379→   379→        title={title}
   380→   380→        onTitleChange={setTitle}
   381→   381→        status={status}
   382→   382→        onUndo={() => editor?.chain().focus().undo().run()}
   383→   383→        onRedo={() => editor?.chain().focus().redo().run()}
   384→   384→        onOpenSearch={() => setSearchOpen(true)}
   385→   385→        onOpenShortcuts={() => setShortcutsOpen(true)}
   386→   386→        onLoadDemo={loadDemo}
   387→   387→      />
   388→   388→
   389→   389→      <main className="relative flex-1 overflow-y-auto" aria-label="Document">
   390→   390→        {/* Loading skeleton */}
   391→   391→        {!ready && (
   392→   392→          <div className="pointer-events-none absolute inset-x-0 top-10 z-10 mx-auto max-w-[720px] px-6">
   393→   393→            <EditorSkeleton />
   394→   394→          </div>
   395→   395→        )}
   396→   396→
   397→   397→        <div ref={contentRef} className="relative mx-auto w-full max-w-[880px] px-5 pb-24 pt-12 md:px-20 lg:px-24">
   398→   398→          <div
   399→   399→            data-editor-column
   400→   400→            className="rich-editor mx-auto max-w-[720px]"
   401→   401→            onContextMenu={(e) => {
   402→   402→              if (!editor) return;
   403→   403→              const coords = editor.view.posAtCoords({ left: e.clientX, top: e.clientY });
   404→   404→              if (!coords) return;
   405→   405→              e.preventDefault();
   406→   406→              setContextMenu({ x: e.clientX, y: e.clientY, pos: coords.pos });
   407→   407→            }}
   408→   408→          >
   409→   409→            {editor && (
   410→   410→              <>
   411→   411→                <EditorContent editor={editor} />
   412→   412→                <FloatingToolbar editor={editor} />
   413→   413→                <TableMenu editor={editor} />
   414→   414→              </>
   415→   415→            )}
   416→   416→          </div>
   417→   417→
   418→   418→          {editor && ready && <BlockHandle editor={editor} contentRef={contentRef} />}
   419→   419→          {editor && ready && <EmptyStateHint editor={editor} onLoadDemo={loadDemo} />}
   420→   420→        </div>
   421→   421→      </main>
   422→   422→
   423→   423→      <StatusBar editor={editor} status={status} savedAt={savedAt} />
   424→   424→
   425→   425→      {/* ------------------------------ overlays ------------------------------ */}
   426→   426→      {editor && <SlashMenu />}
   427→   427→      {editor && <AtMenu editor={editor} />}
   428→   428→      {editor && <EmojiMenu />}
   429→   429→      {editor && <SearchPanel editor={editor} open={searchOpen} onClose={() => setSearchOpen(false)} />}
   430→   430→      {editor && (
   431→   431→        <ContextMenu
   432→   432→          editor={editor}
   433→   433→          state={contextMenu}
   434→   434→          onClose={() => setContextMenu(null)}
   435→   435→        />
   436→   436→      )}
   437→   437→      {editor && <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />}
   438→   438→
   439→   439→      {editor && emojiAnchor && (
   440→   440→        <EmojiPicker
   441→   441→          anchorRect={emojiAnchor}
   442→   442→          onPick={(emoji) => {
   443→   443→            editor.chain().focus().insertContent(emoji).run();
   444→   444→            setEmojiAnchor(null);
   445→   445→          }}
   446→   446→          onClose={() => setEmojiAnchor(null)}
   447→   447→        />
   448→   448→      )}
   449→   449→
   450→   450→      <EmbedDialog
   451→   451→        open={embedOpen}
   452→   452→        onOpenChange={setEmbedOpen}
   453→   453→        onSubmit={(url) => {
   454→   454→          const normalized = normalizeEmbedUrl(url);
   455→   455→          if (!normalized) {
   456→   456→            toast.error("That doesn't look like a valid embed URL");
   457→   457→            return;
   458→   458→          }
   459→   459→          editor?.chain().focus().insertEmbed(normalized).run();
   460→   460→          setEmbedOpen(false);
   461→   461→          toast.success("Embed added");
   462→   462→        }}
   463→   463→      />

   703→   703→function relativeTime(date: Date): string {
   704→   704→  const seconds = Math.round((Date.now() - date.getTime()) / 1000);
   705→   705→  if (seconds < 10) return "just now";
   706→   706→  if (seconds < 60) return `${seconds}s ago`;
   707→   707→  const minutes = Math.round(seconds / 60);
   708→   708→  if (minutes < 60) return `${minutes}m ago`;
   709→   709→  const hours = Math.round(minutes / 60);
   710→   710→  if (hours < 24) return `${hours}h ago`;
   711→   711→  return date.toLocaleDateString();
   712→   712→}
