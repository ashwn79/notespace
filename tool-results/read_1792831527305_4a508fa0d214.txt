    54→    54→/**
    55→    55→ * EditorPage — the full application shell:
    56→    56→ * header (title + tools) · scrollable document surface · status footer.
    57→    57→ */
    58→    58→
    59→    59→type UploadKind = "image" | "video" | "file";
    60→    60→
    61→    61→export default function EditorPage() {
    62→    62→  const [ready, setReady] = React.useState(false);
    63→    63→  const [title, setTitle] = React.useState("");
    64→    64→  const [searchOpen, setSearchOpen] = React.useState(false);
    65→    65→  const [shortcutsOpen, setShortcutsOpen] = React.useState(false);
    66→    66→  const [emojiAnchor, setEmojiAnchor] = React.useState<DOMRect | null>(null);
    67→    67→  const [embedOpen, setEmbedOpen] = React.useState(false);
    68→    68→  const [contextMenu, setContextMenu] = React.useState<ContextMenuState | null>(null);
    69→    69→
    70→    70→  const contentRef = React.useRef<HTMLDivElement>(null);
    71→    71→  const fileInputRef = React.useRef<HTMLInputElement>(null);
    72→    72→  const uploadKindRef = React.useRef<UploadKind>("image");
    73→    73→  const dropInsertPos = React.useRef<number | undefined>(undefined);
    74→    74→
    75→    75→  /* ------------------------------------------------ helpers (slash actions) */
    76→    76→  const requestUpload = React.useCallback((kind: UploadKind) => {
    77→    77→    uploadKindRef.current = kind;
    78→    78→    const input = fileInputRef.current;
    79→    79→    if (input) {
    80→    80→      input.accept =
    81→    81→        kind === "image" ? "image/*" : kind === "video" ? "video/*" : "*/*";
    82→    82→      input.click();
    83→    83→    }
    84→    84→  }, []);
    85→    85→
    86→    86→  const insertUpload = React.useCallback(
    87→    87→    (
    88→    88→      editor: Editor,
    89→    89→      result: { url: string; name: string; size: number; type: string },
    90→    90→      kind: UploadKind,
    91→    91→      replacePos?: number,
    92→    92→    ) => {
    93→    93→      const node: JSONContent =
    94→    94→        kind === "image"
    95→    95→          ? { type: "imageBlock", attrs: { src: result.url, alt: result.name, width: 100, align: "center" } }
    96→    96→          : kind === "video"
    97→    97→            ? { type: "videoBlock", attrs: { src: result.url, name: result.name } }
    98→    98→            : { type: "fileBlock", attrs: { src: result.url, name: result.name, size: result.size, type: result.type } };
    99→    99→      // A selected media placeholder is REPLACED by the finished block.
   100→   100→      if (replacePos !== undefined) {
   101→   101→        const target = editor.state.doc.nodeAt(replacePos);
   102→   102→        const to = target ? replacePos + target.nodeSize : replacePos + 1;
   103→   103→        editor.chain().focus().insertContentAt({ from: replacePos, to }, node).run();
   104→   104→        return;
   105→   105→      }
   106→   106→      editor
   107→   107→        .chain()
   108→   108→        .focus()
   109→   109→        .insertContentAt(dropInsertPos.current ?? editor.state.selection.to, node)
   110→   110→        .run();
   111→   111→      dropInsertPos.current = undefined;
   112→   112→    },
   113→   113→    [],
   114→   114→  );
   115→   115→
   116→   116→  const handleFiles = React.useCallback(
   117→   117→    async (files: File[], editorInstance: Editor, insertPos?: number, replacePos?: number) => {
   118→   118→      dropInsertPos.current = insertPos;
   119→   119→      for (const file of files) {
   120→   120→        const loadingToast = toast.loading(`Uploading ${file.name}…`, {
   121→   121→          description: formatBytes(file.size),
   122→   122→        });
   123→   123→        try {
   124→   124→          const result = await uploadFile(file);
   125→   125→          toast.dismiss(loadingToast);
   126→   126→          // Choose the block type from the actual mime type (handles
   127→   127→          // drag-drop where the slash-menu kind is not authoritative).
   128→   128→          const kind: UploadKind = file.type.startsWith("image")
   129→   129→            ? "image"
   130→   130→            : file.type.startsWith("video")
   131→   131→              ? "video"
   132→   132→              : "file";
   133→   133→          insertUpload(editorInstance, result, kind, replacePos);
   134→   134→          toast.success(`${file.name} added`);
   135→   135→        } catch (error) {
   136→   136→          toast.dismiss(loadingToast);
   137→   137→          toast.error(`Upload failed: ${error instanceof Error ? error.message : "unknown error"}`);
   138→   138→        }
   139→   139→      }
   140→   140→    },
   141→   141→    [insertUpload],
   142→   142→  );
   143→   143→
   144→   144→  /** If the current selection is a media placeholder, its document position. */
   145→   145→  const selectedPlaceholderPos = React.useCallback((editorInstance: Editor): number | undefined => {
   146→   146→    const { selection } = editorInstance.state;
   147→   147→    if (selection instanceof NodeSelection && selection.node.type.name === "mediaPlaceholder") {
   148→   148→      return selection.from;
   149→   149→    }
   150→   150→    return undefined;
   151→   151→  }, []);
   152→   152→
   153→   153→  /** Commit a URL into the selected media placeholder (paste-a-link UX). */
   154→   154→  const commitPlaceholderUrl = React.useCallback(
   155→   155→    (editorInstance: Editor, pos: number, rawUrl: string): boolean => {
   156→   156→      const node = editorInstance.state.doc.nodeAt(pos);
   157→   157→      if (!node || node.type.name !== "mediaPlaceholder") return false;
   158→   158→      const rawKind = node.attrs.kind;
   159→   159→      const kind: MediaKind = isMediaKind(rawKind) ? rawKind : "image";
   160→   160→      const url = /^https?:\/\//i.test(rawUrl.trim()) ? rawUrl.trim() : `https://${rawUrl.trim()}`;
   161→   161→      let name = "";
   162→   162→      try {
   163→   163→        const parsed = new URL(url);
   164→   164→        name =
   165→   165→          decodeURIComponent(parsed.pathname.split("/").filter(Boolean).pop() ?? "") || parsed.hostname;
   166→   166→      } catch {
   167→   167→        return false;
   168→   168→      }
   169→   169→      editorInstance
   170→   170→        .chain()
   171→   171→        .focus()
   172→   172→        .insertContentAt(
   173→   173→          { from: pos, to: pos + node.nodeSize },
   174→   174→          mediaNodeFor(kind, {
   175→   175→            url,
   176→   176→            name,
   177→   177→            size: 0,
   178→   178→            type: kind === "image" ? "image/remote" : kind === "video" ? "video/remote" : guessTypeFromUrl(url),
   179→   179→          }),
   180→   180→        )
   181→   181→        .run();
   182→   182→      return true;
   183→   183→    },
   184→   184→    [],
   185→   185→  );
   186→   186→
   187→   187→  /* ------------------------------------------------ slash action helpers */
   188→   188→  const requestEmoji = React.useCallback((editorInstance: Editor) => {
   189→   189→    const coords = editorInstance.view.coordsAtPos(editorInstance.state.selection.to);
   190→   190→    setEmojiAnchor(new DOMRect(coords.left, coords.top, 0, 0));
   191→   191→  }, []);
   192→   192→
   193→   193→  const requestEmbed = React.useCallback(() => setEmbedOpen(true), []);
   194→   194→
   195→   195→  const requestMath = React.useCallback((editorInstance: Editor, inline: boolean) => {
   196→   196→    if (inline) editorInstance.chain().focus().insertMathInline("").run();
   197→   197→    else editorInstance.chain().focus().insertMathBlock("").run();
   198→   198→  }, []);
   199→   199→
   200→   200→  // Stable across renders — safe to capture once inside useEditor options.
   201→   201→  const slashHelpers = React.useMemo<SlashActionHelpers>(
   202→   202→    () => ({ requestUpload, requestEmoji, requestEmbed, requestMath }),
   203→   203→    [requestUpload, requestEmoji, requestEmbed, requestMath],
   204→   204→  );
   205→   205→
   206→   206→  /* ------------------------------------------------ @ menu dialog helpers */
   207→   207→  const datePickCb = React.useRef<((iso: string) => void) | null>(null);
   208→   208→  const personCb = React.useRef<((name: string) => void) | null>(null);
   209→   209→  const [datePickOpen, setDatePickOpen] = React.useState(false);
   210→   210→  const [personOpen, setPersonOpen] = React.useState(false);
   211→   211→
   212→   212→  const mentionHelpers = React.useMemo<AtActionHelpers>(
   213→   213→    () => ({
   214→   214→      requestDatePick: (onPick) => {
   215→   215→        datePickCb.current = onPick;
   216→   216→        setDatePickOpen(true);
   217→   217→      },
   218→   218→      requestPersonInput: (onSubmit) => {
   219→   219→        personCb.current = onSubmit;
   220→   220→        setPersonOpen(true);
   221→   221→      },
   222→   222→    }),
   223→   223→    [],
   224→   224→  );
   225→   225→
   226→   226→  // The editor instance lands in a ref *after* creation so paste/drop
   227→   227→  // handlers (created before the editor exists) can reach it safely.
   228→   228→  const editorRef = React.useRef<Editor | null>(null);
   229→   229→
   230→   230→  /* ------------------------------------------------ editor instance */
   231→   231→  const editor = useEditor(
   232→   232→    {
   233→   233→      extensions: buildExtensions({
   234→   234→        getActionContext: () => slashHelpers,
   235→   235→        getMentionHelpers: () => mentionHelpers,
   236→   236→      }),
   237→   237→      content: { type: "doc", content: [{ type: "paragraph" }] },
   238→   238→      immediatelyRender: false,
   239→   239→      autofocus: false,
   240→   240→      editorProps: {
   241→   241→        attributes: {
   242→   242→          class: "tiptap",
   243→   243→          "aria-label": "Document editor",
   244→   244→          role: "textbox",
   245→   245→        },
   246→   246→        handlePaste: (_view, event) => {
   247→   247→          const editorInstance = editorRef.current;
   248→   248→          if (!editorInstance) return false;
   249→   249→          const clipboard = event.clipboardData;
   250→   250→          const files = Array.from(clipboard?.files ?? []);
   251→   251→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   252→   252→
   253→   253→          // Pasting a URL onto a selected placeholder commits it directly.
   254→   254→          if (placeholderPos !== undefined && !files.length) {
   255→   255→            const text = (clipboard?.getData("text/plain") ?? "").trim();
   256→   256→            if (/^(https?:\/\/|www\.)\S+$/i.test(text) && commitPlaceholderUrl(editorInstance, placeholderPos, text)) {
   257→   257→              return true;
   258→   258→            }
   259→   259→          }
   260→   260→
   261→   261→          if (!files.length) return false;
   262→   262→          // Screenshots & copied files — pick the block kind by mime type.
   263→   263→          for (const file of files) {
   264→   264→            void handleFiles([file], editorInstance, undefined, placeholderPos);
   265→   265→          }
   266→   266→          return true;
   267→   267→        },
   268→   268→        handleDrop: (view, event, _slice, moved) => {
   269→   269→          const editorInstance = editorRef.current;
   270→   270→          if (moved || !editorInstance) return false;
   271→   271→          // Drops on a media placeholder card are handled by the card itself.
   272→   272→          if ((event.target as HTMLElement)?.closest?.("[data-media-placeholder-ui]")) return false;
   273→   273→          const files = Array.from(event.dataTransfer?.files ?? []);
   274→   274→          if (!files.length) return false;
   275→   275→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   276→   276→          const coords = view.posAtCoords({ left: event.clientX, top: event.clientY });
   277→   277→          for (const file of files) {
   278→   278→            void handleFiles([file], editorInstance, coords?.pos, placeholderPos);
   279→   279→          }
   280→   280→          return true;
   281→   281→        },
   282→   282→      },
   283→   283→    },
   284→   284→    [],
   285→   285→  );
   286→   286→
   287→   287→  /* ------------------------------------------------ keep editorRef fresh */
   288→   288→  React.useEffect(() => {
   289→   289→    editorRef.current = editor;
   290→   290→    // Debug handle for browser console testing (dev only).
   291→   291→    if (process.env.NODE_ENV !== "production") {
   292→   292→      (window as unknown as { __editor?: Editor }).__editor = editor;
   293→   293→    }
   294→   294→  }, [editor]);
   295→   295→
   296→   296→  /* ------------------------------------------------ load document */
   297→   297→  React.useEffect(() => {
   298→   298→    if (!editor) return;
   299→   299→    let cancelled = false;
   300→   300→    fetchDocument()
   301→   301→      .then((doc) => {
   302→   302→        if (cancelled) return;
   303→   303→        setTitle(doc.title);
   304→   304→        if (doc.content && typeof doc.content === "object" && "type" in (doc.content as object)) {
   305→   305→          editor.commands.setContent(doc.content as JSONContent, { emitUpdate: false });
   306→   306→        }
   307→   307→        setReady(true);
   308→   308→      })
   309→   309→      .catch((error) => {
   310→   310→        if (cancelled) return;
   311→   311→        console.error("[editor] failed to load document", error);
   312→   312→        toast.error("Couldn't load your document", {
   313→   313→          description: "You can keep writing — changes will retry on save.",
   314→   314→        });
   315→   315→        setReady(true);
   316→   316→      });
   317→   317→    return () => {
   318→   318→      cancelled = true;
   319→   319→    };
   320→   320→  }, [editor]);
   321→   321→
   322→   322→  /* ------------------------------------------------ autosave */
   323→   323→  const { status, savedAt, saveNow } = useAutosave({ editor, title, ready });
   324→   324→
   325→   325→  /* ------------------------------------------------ global shortcuts */
   326→   326→  React.useEffect(() => {
   327→   327→    const onKey = (e: KeyboardEvent) => {
   328→   328→      const mod = e.metaKey || e.ctrlKey;
   329→   329→      if (mod && e.key.toLowerCase() === "f") {
   330→   330→        e.preventDefault();
   331→   331→        setSearchOpen(true);
   332→   332→      }
   333→   333→      if (mod && e.key === "/") {
   334→   334→        e.preventDefault();
   335→   335→        setShortcutsOpen((v) => !v);
   336→   336→      }
   337→   337→    };
   338→   338→    window.addEventListener("keydown", onKey);
   339→   339→    return () => window.removeEventListener("keydown", onKey);
   340→   340→  }, []);
   341→   341→
   342→   342→  /* ------------------------------------------------ future-ready stubs */
   343→   343→  React.useEffect(() => {
   344→   344→    const onFutureFeature = (e: Event) => {
   345→   345→      const detail = (e as CustomEvent<string>).detail ?? "This feature";
   346→   346→      toast(`${detail} — coming soon`, {
   347→   347→        description: "The extension registry is ready for it.",
   348→   348→      });
   349→   349→    };
   350→   350→    window.addEventListener("aurora:future-feature", onFutureFeature);
   351→   351→    return () => window.removeEventListener("aurora:future-feature", onFutureFeature);
   352→   352→  }, []);
   353→   353→
   354→   354→  /* ------------------------------------------------ demo & misc actions */
   355→   355→  const loadDemo = React.useCallback(() => {
   356→   356→    if (!editor) return;
   357→   357→    if (editor.isEmpty) {
   358→   358→      editor.commands.setContent(DEMO_CONTENT, { emitUpdate: true });
   359→   359→    } else {
   360→   360→      editor
   361→   361→        .chain()
   362→   362→        .focus("end")
   363→   363→        .insertContentAt(editor.state.doc.content.size, DEMO_CONTENT.content)
   364→   364→        .run();
   365→   365→    }
   366→   366→    toast.success("Demo document loaded", {
   367→   367→      description: "Hover blocks, press / anywhere, select text — explore!",
   368→   368→    });
   369→   369→  }, [editor]);
   370→   370→
   371→   371→  return (
   372→   372→    <div className="flex h-screen flex-col overflow-hidden bg-background">
   373→   373→      <Header
   374→   374→        editor={editor}
   375→   375→        title={title}
   376→   376→        onTitleChange={setTitle}
   377→   377→        status={status}
   378→   378→        onUndo={() => editor?.chain().focus().undo().run()}
   379→   379→        onRedo={() => editor?.chain().focus().redo().run()}
   380→   380→        onOpenSearch={() => setSearchOpen(true)}
   381→   381→        onOpenShortcuts={() => setShortcutsOpen(true)}
   382→   382→        onLoadDemo={loadDemo}
   383→   383→      />
   384→   384→
   385→   385→      <main className="relative flex-1 overflow-y-auto" aria-label="Document">
   386→   386→        {/* Loading skeleton */}
   387→   387→        {!ready && (
   388→   388→          <div className="pointer-events-none absolute inset-x-0 top-10 z-10 mx-auto max-w-[720px] px-6">
   389→   389→            <EditorSkeleton />
   390→   390→          </div>
   391→   391→        )}
   392→   392→
   393→   393→        <div ref={contentRef} className="relative mx-auto w-full max-w-[880px] px-5 pb-24 pt-12 md:px-20 lg:px-24">
   394→   394→          <div
   395→   395→            data-editor-column
   396→   396→            className="rich-editor mx-auto max-w-[720px]"
   397→   397→            onContextMenu={(e) => {
   398→   398→              if (!editor) return;
   399→   399→              const coords = editor.view.posAtCoords({ left: e.clientX, top: e.clientY });
   400→   400→              if (!coords) return;
   401→   401→              e.preventDefault();
   402→   402→              setContextMenu({ x: e.clientX, y: e.clientY, pos: coords.pos });
   403→   403→            }}
   404→   404→          >
   405→   405→            {editor && (
   406→   406→              <>
   407→   407→                <EditorContent editor={editor} />
   408→   408→                <FloatingToolbar editor={editor} />
   409→   409→                <TableMenu editor={editor} />
   410→   410→              </>
   411→   411→            )}
   412→   412→          </div>
   413→   413→
   414→   414→          {editor && ready && <BlockHandle editor={editor} contentRef={contentRef} />}
   415→   415→          {editor && ready && <EmptyStateHint editor={editor} onLoadDemo={loadDemo} />}
   416→   416→        </div>
   417→   417→      </main>
   418→   418→
   419→   419→      <StatusBar editor={editor} status={status} savedAt={savedAt} />
   420→   420→
   421→   421→      {/* ------------------------------ overlays ------------------------------ */}
   422→   422→      {editor && <SlashMenu />}
   423→   423→      {editor && <AtMenu />}
   424→   424→      {editor && <EmojiMenu />}
   425→   425→      {editor && <SearchPanel editor={editor} open={searchOpen} onClose={() => setSearchOpen(false)} />}
   426→   426→      {editor && (
   427→   427→        <ContextMenu
   428→   428→          editor={editor}
   429→   429→          state={contextMenu}
   430→   430→          onClose={() => setContextMenu(null)}
   431→   431→        />
   432→   432→      )}
   433→   433→      {editor && <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />}
   434→   434→
   435→   435→      {editor && emojiAnchor && (
   436→   436→        <EmojiPicker
   437→   437→          anchorRect={emojiAnchor}
   438→   438→          onPick={(emoji) => {
   439→   439→            editor.chain().focus().insertContent(emoji).run();
   440→   440→            setEmojiAnchor(null);
   441→   441→          }}
   442→   442→          onClose={() => setEmojiAnchor(null)}
   443→   443→        />
   444→   444→      )}
   445→   445→
   446→   446→      <EmbedDialog
   447→   447→        open={embedOpen}
   448→   448→        onOpenChange={setEmbedOpen}
   449→   449→        onSubmit={(url) => {
   450→   450→          const normalized = normalizeEmbedUrl(url);
   451→   451→          if (!normalized) {
   452→   452→            toast.error("That doesn't look like a valid embed URL");
   453→   453→            return;
   454→   454→          }
   455→   455→          editor?.chain().focus().insertEmbed(normalized).run();
   456→   456→          setEmbedOpen(false);
   457→   457→          toast.success("Embed added");
   458→   458→        }}
   459→   459→      />
   460→   460→
   461→   461→      <DatePickDialog
   462→   462→        open={datePickOpen}
   463→   463→        onOpenChange={setDatePickOpen}
   464→   464→        onPick={(iso) => {
   465→   465→          datePickCb.current?.(iso);
   466→   466→          datePickCb.current = null;
   467→   467→        }}

   706→   706→function relativeTime(date: Date): string {
   707→   707→  const seconds = Math.round((Date.now() - date.getTime()) / 1000);
   708→   708→  if (seconds < 10) return "just now";
   709→   709→  if (seconds < 60) return `${seconds}s ago`;
   710→   710→  const minutes = Math.round(seconds / 60);
   711→   711→  if (minutes < 60) return `${minutes}m ago`;
   712→   712→  const hours = Math.round(minutes / 60);
   713→   713→  if (hours < 24) return `${hours}h ago`;
   714→   714→  return date.toLocaleDateString();
   715→   715→}
