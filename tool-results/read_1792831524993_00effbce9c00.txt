    54→/**
    55→ * EditorPage — the full application shell:
    56→ * header (title + tools) · scrollable document surface · status footer.
    57→ */
    58→
    59→type UploadKind = "image" | "video" | "file";
    60→
    61→export default function EditorPage() {
    62→  const [ready, setReady] = React.useState(false);
    63→  const [title, setTitle] = React.useState("");
    64→  const [searchOpen, setSearchOpen] = React.useState(false);
    65→  const [shortcutsOpen, setShortcutsOpen] = React.useState(false);
    66→  const [emojiAnchor, setEmojiAnchor] = React.useState<DOMRect | null>(null);
    67→  const [embedOpen, setEmbedOpen] = React.useState(false);
    68→  const [contextMenu, setContextMenu] = React.useState<ContextMenuState | null>(null);
    69→
    70→  const contentRef = React.useRef<HTMLDivElement>(null);
    71→  const fileInputRef = React.useRef<HTMLInputElement>(null);
    72→  const uploadKindRef = React.useRef<UploadKind>("image");
    73→  const dropInsertPos = React.useRef<number | undefined>(undefined);
    74→
    75→  /* ------------------------------------------------ helpers (slash actions) */
    76→  const requestUpload = React.useCallback((kind: UploadKind) => {
    77→    uploadKindRef.current = kind;
    78→    const input = fileInputRef.current;
    79→    if (input) {
    80→      input.accept =
    81→        kind === "image" ? "image/*" : kind === "video" ? "video/*" : "*/*";
    82→      input.click();
    83→    }
    84→  }, []);
    85→
    86→  const insertUpload = React.useCallback(
    87→    (
    88→      editor: Editor,
    89→      result: { url: string; name: string; size: number; type: string },
    90→      kind: UploadKind,
    91→      replacePos?: number,
    92→    ) => {
    93→      const node: JSONContent =
    94→        kind === "image"
    95→          ? { type: "imageBlock", attrs: { src: result.url, alt: result.name, width: 100, align: "center" } }
    96→          : kind === "video"
    97→            ? { type: "videoBlock", attrs: { src: result.url, name: result.name } }
    98→            : { type: "fileBlock", attrs: { src: result.url, name: result.name, size: result.size, type: result.type } };
    99→      // A selected media placeholder is REPLACED by the finished block.
   100→      if (replacePos !== undefined) {
   101→        const target = editor.state.doc.nodeAt(replacePos);
   102→        const to = target ? replacePos + target.nodeSize : replacePos + 1;
   103→        editor.chain().focus().insertContentAt({ from: replacePos, to }, node).run();
   104→        return;
   105→      }
   106→      editor
   107→        .chain()
   108→        .focus()
   109→        .insertContentAt(dropInsertPos.current ?? editor.state.selection.to, node)
   110→        .run();
   111→      dropInsertPos.current = undefined;
   112→    },
   113→    [],
   114→  );
   115→
   116→  const handleFiles = React.useCallback(
   117→    async (files: File[], editorInstance: Editor, insertPos?: number, replacePos?: number) => {
   118→      dropInsertPos.current = insertPos;
   119→      for (const file of files) {
   120→        const loadingToast = toast.loading(`Uploading ${file.name}…`, {
   121→          description: formatBytes(file.size),
   122→        });
   123→        try {
   124→          const result = await uploadFile(file);
   125→          toast.dismiss(loadingToast);
   126→          // Choose the block type from the actual mime type (handles
   127→          // drag-drop where the slash-menu kind is not authoritative).
   128→          const kind: UploadKind = file.type.startsWith("image")
   129→            ? "image"
   130→            : file.type.startsWith("video")
   131→              ? "video"
   132→              : "file";
   133→          insertUpload(editorInstance, result, kind, replacePos);
   134→          toast.success(`${file.name} added`);
   135→        } catch (error) {
   136→          toast.dismiss(loadingToast);
   137→          toast.error(`Upload failed: ${error instanceof Error ? error.message : "unknown error"}`);
   138→        }
   139→      }
   140→    },
   141→    [insertUpload],
   142→  );
   143→
   144→  /** If the current selection is a media placeholder, its document position. */
   145→  const selectedPlaceholderPos = React.useCallback((editorInstance: Editor): number | undefined => {
   146→    const { selection } = editorInstance.state;
   147→    if (selection instanceof NodeSelection && selection.node.type.name === "mediaPlaceholder") {
   148→      return selection.from;
   149→    }
   150→    return undefined;
   151→  }, []);
   152→
   153→  /** Commit a URL into the selected media placeholder (paste-a-link UX). */
   154→  const commitPlaceholderUrl = React.useCallback(
   155→    (editorInstance: Editor, pos: number, rawUrl: string): boolean => {
   156→      const node = editorInstance.state.doc.nodeAt(pos);
   157→      if (!node || node.type.name !== "mediaPlaceholder") return false;
   158→      const rawKind = node.attrs.kind;
   159→      const kind: MediaKind = isMediaKind(rawKind) ? rawKind : "image";
   160→      const url = /^https?:\/\//i.test(rawUrl.trim()) ? rawUrl.trim() : `https://${rawUrl.trim()}`;
   161→      let name = "";
   162→      try {
   163→        const parsed = new URL(url);
   164→        name =
   165→          decodeURIComponent(parsed.pathname.split("/").filter(Boolean).pop() ?? "") || parsed.hostname;
   166→      } catch {
   167→        return false;
   168→      }
   169→      editorInstance
   170→        .chain()
   171→        .focus()
   172→        .insertContentAt(
   173→          { from: pos, to: pos + node.nodeSize },
   174→          mediaNodeFor(kind, {
   175→            url,
   176→            name,
   177→            size: 0,
   178→            type: kind === "image" ? "image/remote" : kind === "video" ? "video/remote" : guessTypeFromUrl(url),
   179→          }),
   180→        )
   181→        .run();
   182→      return true;
   183→    },
   184→    [],
   185→  );
   186→
   187→  /* ------------------------------------------------ slash action helpers */
   188→  const requestEmoji = React.useCallback((editorInstance: Editor) => {
   189→    const coords = editorInstance.view.coordsAtPos(editorInstance.state.selection.to);
   190→    setEmojiAnchor(new DOMRect(coords.left, coords.top, 0, 0));
   191→  }, []);
   192→
   193→  const requestEmbed = React.useCallback(() => setEmbedOpen(true), []);
   194→
   195→  const requestMath = React.useCallback((editorInstance: Editor, inline: boolean) => {
   196→    if (inline) editorInstance.chain().focus().insertMathInline("").run();
   197→    else editorInstance.chain().focus().insertMathBlock("").run();
   198→  }, []);
   199→
   200→  // Stable across renders — safe to capture once inside useEditor options.
   201→  const slashHelpers = React.useMemo<SlashActionHelpers>(
   202→    () => ({ requestUpload, requestEmoji, requestEmbed, requestMath }),
   203→    [requestUpload, requestEmoji, requestEmbed, requestMath],
   204→  );
   205→
   206→  /* ------------------------------------------------ @ menu dialog helpers */
   207→  const datePickCb = React.useRef<((iso: string) => void) | null>(null);
   208→  const personCb = React.useRef<((name: string) => void) | null>(null);
   209→  const [datePickOpen, setDatePickOpen] = React.useState(false);
   210→  const [personOpen, setPersonOpen] = React.useState(false);
   211→
   212→  const mentionHelpers = React.useMemo<AtActionHelpers>(
   213→    () => ({
   214→      requestDatePick: (onPick) => {
   215→        datePickCb.current = onPick;
   216→        setDatePickOpen(true);
   217→      },
   218→      requestPersonInput: (onSubmit) => {
   219→        personCb.current = onSubmit;
   220→        setPersonOpen(true);
   221→      },
   222→    }),
   223→    [],
   224→  );
   225→
   226→  // The editor instance lands in a ref *after* creation so paste/drop
   227→  // handlers (created before the editor exists) can reach it safely.
   228→  const editorRef = React.useRef<Editor | null>(null);
   229→
   230→  /* ------------------------------------------------ editor instance */
   231→  const editor = useEditor(
   232→    {
   233→      extensions: buildExtensions({
   234→        getActionContext: () => slashHelpers,
   235→        getMentionHelpers: () => mentionHelpers,
   236→      }),
   237→      content: { type: "doc", content: [{ type: "paragraph" }] },
   238→      immediatelyRender: false,
   239→      autofocus: false,
   240→      editorProps: {
   241→        attributes: {
   242→          class: "tiptap",
   243→          "aria-label": "Document editor",
   244→          role: "textbox",
   245→        },
   246→        handlePaste: (_view, event) => {
   247→          const editorInstance = editorRef.current;
   248→          if (!editorInstance) return false;
   249→          const clipboard = event.clipboardData;
   250→          const files = Array.from(clipboard?.files ?? []);
   251→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   252→
   253→          // Pasting a URL onto a selected placeholder commits it directly.
   254→          if (placeholderPos !== undefined && !files.length) {
   255→            const text = (clipboard?.getData("text/plain") ?? "").trim();
   256→            if (/^(https?:\/\/|www\.)\S+$/i.test(text) && commitPlaceholderUrl(editorInstance, placeholderPos, text)) {
   257→              return true;
   258→            }
   259→          }
   260→
   261→          if (!files.length) return false;
   262→          // Screenshots & copied files — pick the block kind by mime type.
   263→          for (const file of files) {
   264→            void handleFiles([file], editorInstance, undefined, placeholderPos);
   265→          }
   266→          return true;
   267→        },
   268→        handleDrop: (view, event, _slice, moved) => {
   269→          const editorInstance = editorRef.current;
   270→          if (moved || !editorInstance) return false;
   271→          // Drops on a media placeholder card are handled by the card itself.
   272→          if ((event.target as HTMLElement)?.closest?.("[data-media-placeholder-ui]")) return false;
   273→          const files = Array.from(event.dataTransfer?.files ?? []);
   274→          if (!files.length) return false;
   275→          const placeholderPos = selectedPlaceholderPos(editorInstance);
   276→          const coords = view.posAtCoords({ left: event.clientX, top: event.clientY });
   277→          for (const file of files) {
   278→            void handleFiles([file], editorInstance, coords?.pos, placeholderPos);
   279→          }
   280→          return true;
   281→        },
   282→      },
   283→    },
   284→    [],
   285→  );
   286→
   287→  /* ------------------------------------------------ keep editorRef fresh */
   288→  React.useEffect(() => {
   289→    editorRef.current = editor;
   290→    // Debug handle for browser console testing (dev only).
   291→    if (process.env.NODE_ENV !== "production") {
   292→      (window as unknown as { __editor?: Editor }).__editor = editor;
   293→    }
   294→  }, [editor]);
   295→
   296→  /* ------------------------------------------------ load document */
   297→  React.useEffect(() => {
   298→    if (!editor) return;
   299→    let cancelled = false;
   300→    fetchDocument()
   301→      .then((doc) => {
   302→        if (cancelled) return;
   303→        setTitle(doc.title);
   304→        if (doc.content && typeof doc.content === "object" && "type" in (doc.content as object)) {
   305→          editor.commands.setContent(doc.content as JSONContent, { emitUpdate: false });
   306→        }
   307→        setReady(true);
   308→      })
   309→      .catch((error) => {
   310→        if (cancelled) return;
   311→        console.error("[editor] failed to load document", error);
   312→        toast.error("Couldn't load your document", {
   313→          description: "You can keep writing — changes will retry on save.",
   314→        });
   315→        setReady(true);
   316→      });
   317→    return () => {
   318→      cancelled = true;
   319→    };
   320→  }, [editor]);
   321→
   322→  /* ------------------------------------------------ autosave */
   323→  const { status, savedAt, saveNow } = useAutosave({ editor, title, ready });
   324→
   325→  /* ------------------------------------------------ global shortcuts */
   326→  React.useEffect(() => {
   327→    const onKey = (e: KeyboardEvent) => {
   328→      const mod = e.metaKey || e.ctrlKey;
   329→      if (mod && e.key.toLowerCase() === "f") {
   330→        e.preventDefault();
   331→        setSearchOpen(true);
   332→      }
   333→      if (mod && e.key === "/") {
   334→        e.preventDefault();
   335→        setShortcutsOpen((v) => !v);
   336→      }
   337→    };
   338→    window.addEventListener("keydown", onKey);
   339→    return () => window.removeEventListener("keydown", onKey);
   340→  }, []);
   341→
   342→  /* ------------------------------------------------ future-ready stubs */
   343→  React.useEffect(() => {
   344→    const onFutureFeature = (e: Event) => {
   345→      const detail = (e as CustomEvent<string>).detail ?? "This feature";
   346→      toast(`${detail} — coming soon`, {
   347→        description: "The extension registry is ready for it.",
   348→      });
   349→    };
   350→    window.addEventListener("aurora:future-feature", onFutureFeature);
   351→    return () => window.removeEventListener("aurora:future-feature", onFutureFeature);
   352→  }, []);
   353→
   354→  /* ------------------------------------------------ demo & misc actions */
   355→  const loadDemo = React.useCallback(() => {
   356→    if (!editor) return;
   357→    if (editor.isEmpty) {
   358→      editor.commands.setContent(DEMO_CONTENT, { emitUpdate: true });
   359→    } else {
   360→      editor
   361→        .chain()
   362→        .focus("end")
   363→        .insertContentAt(editor.state.doc.content.size, DEMO_CONTENT.content)
   364→        .run();
   365→    }
   366→    toast.success("Demo document loaded", {
   367→      description: "Hover blocks, press / anywhere, select text — explore!",
   368→    });
   369→  }, [editor]);
   370→
   371→  return (
   372→    <div className="flex h-screen flex-col overflow-hidden bg-background">
   373→      <Header
   374→        editor={editor}
   375→        title={title}
   376→        onTitleChange={setTitle}
   377→        status={status}
   378→        onUndo={() => editor?.chain().focus().undo().run()}
   379→        onRedo={() => editor?.chain().focus().redo().run()}
   380→        onOpenSearch={() => setSearchOpen(true)}
   381→        onOpenShortcuts={() => setShortcutsOpen(true)}
   382→        onLoadDemo={loadDemo}
   383→      />
   384→
   385→      <main className="relative flex-1 overflow-y-auto" aria-label="Document">
   386→        {/* Loading skeleton */}
   387→        {!ready && (
   388→          <div className="pointer-events-none absolute inset-x-0 top-10 z-10 mx-auto max-w-[720px] px-6">
   389→            <EditorSkeleton />
   390→          </div>
   391→        )}
   392→
   393→        <div ref={contentRef} className="relative mx-auto w-full max-w-[880px] px-5 pb-24 pt-12 md:px-20 lg:px-24">
   394→          <div
   395→            data-editor-column
   396→            className="rich-editor mx-auto max-w-[720px]"
   397→            onContextMenu={(e) => {
   398→              if (!editor) return;
   399→              const coords = editor.view.posAtCoords({ left: e.clientX, top: e.clientY });
   400→              if (!coords) return;
   401→              e.preventDefault();
   402→              setContextMenu({ x: e.clientX, y: e.clientY, pos: coords.pos });
   403→            }}
   404→          >
   405→            {editor && (
   406→              <>
   407→                <EditorContent editor={editor} />
   408→                <FloatingToolbar editor={editor} />
   409→                <TableMenu editor={editor} />
   410→              </>
   411→            )}
   412→          </div>
   413→
   414→          {editor && ready && <BlockHandle editor={editor} contentRef={contentRef} />}
   415→          {editor && ready && <EmptyStateHint editor={editor} onLoadDemo={loadDemo} />}
   416→        </div>
   417→      </main>
   418→
   419→      <StatusBar editor={editor} status={status} savedAt={savedAt} />
   420→
   421→      {/* ------------------------------ overlays ------------------------------ */}
   422→      {editor && <SlashMenu />}
   423→      {editor && <AtMenu />}
   424→      {editor && <EmojiMenu />}
   425→      {editor && <SearchPanel editor={editor} open={searchOpen} onClose={() => setSearchOpen(false)} />}
   426→      {editor && (
   427→        <ContextMenu
   428→          editor={editor}
   429→          state={contextMenu}
   430→          onClose={() => setContextMenu(null)}
   431→        />
   432→      )}
   433→      {editor && <ShortcutsDialog open={shortcutsOpen} onOpenChange={setShortcutsOpen} />}
   434→
   435→      {editor && emojiAnchor && (
   436→        <EmojiPicker
   437→          anchorRect={emojiAnchor}
   438→          onPick={(emoji) => {
   439→            editor.chain().focus().insertContent(emoji).run();
   440→            setEmojiAnchor(null);
   441→          }}
   442→          onClose={() => setEmojiAnchor(null)}
   443→        />
   444→      )}
   445→
   446→      <EmbedDialog
   447→        open={embedOpen}
   448→        onOpenChange={setEmbedOpen}
   449→        onSubmit={(url) => {
   450→          const normalized = normalizeEmbedUrl(url);
   451→          if (!normalized) {
   452→            toast.error("That doesn't look like a valid embed URL");
   453→            return;
   454→          }
   455→          editor?.chain().focus().insertEmbed(normalized).run();
   456→          setEmbedOpen(false);
   457→          toast.success("Embed added");
   458→        }}
   459→      />
   460→
   461→      <DatePickDialog
   462→        open={datePickOpen}
   463→        onOpenChange={setDatePickOpen}
   464→        onPick={(iso) => {
   465→          datePickCb.current?.(iso);
   466→          datePickCb.current = null;
   467→        }}
   468→      />
   469→      <PersonDialog
   470→        open={personOpen}
   471→        onOpenChange={setPersonOpen}
   472→        onSubmit={(name) => {
   473→          personCb.current?.(name);
   474→          personCb.current = null;
   475→        }}
   476→      />
   477→
   478→      {/* Hidden file input for uploads */}
   479→      <input
   480→        ref={fileInputRef}
   481→        type="file"
   482→        multiple
   483→        className="hidden"
   484→        onChange={(e) => {
   485→          const files = Array.from(e.target.files ?? []);
   486→          if (files.length && editor) void handleFiles(files, editor);
   487→          e.target.value = "";
   488→        }}
   489→      />
   490→    </div>
   491→  );
   492→}

   494→/* ================================================================ Header */
   495→
   496→function Header({
   497→  editor,
   498→  title,
   499→  onTitleChange,
   500→  status,
   501→  onUndo,
   502→  onRedo,
   503→  onOpenSearch,
   504→  onOpenShortcuts,
   505→  onLoadDemo,
   506→}: {
   507→  editor: Editor | null;
   508→  title: string;
   509→  onTitleChange: (value: string) => void;
   510→  status: SaveStatus;
   511→  onUndo: () => void;
   512→  onRedo: () => void;
   513→  onOpenSearch: () => void;
   514→  onOpenShortcuts: () => void;
   515→  onLoadDemo: () => void;
   516→}) {
   517→  const { theme, setTheme } = useTheme();
   518→  const [mounted, setMounted] = React.useState(false);
   519→  React.useEffect(() => setMounted(true), []);
   520→
   521→  const history = useEditorState({
   522→    editor,
   523→    selector: ({ editor: e }) =>
   524→      e ? { canUndo: e.can().undo(), canRedo: e.can().redo() } : { canUndo: false, canRedo: false },
   525→  });
   526→
   527→  return (
   528→    <TooltipProvider delayDuration={300}>
   529→      <header className="sticky top-0 z-50 shrink-0 border-b border-border/60 bg-background/85 backdrop-blur-xl">
   530→        <div className="mx-auto flex h-[52px] w-full max-w-5xl items-center gap-2 px-3 md:px-5">
   531→          <div
   532→            className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-gradient-to-br from-violet-500 via-purple-500 to-fuchsia-500 text-white shadow-md shadow-violet-500/25"
   533→            aria-hidden
   534→          >
   535→            <Feather size={15} strokeWidth={2.2} />
   536→          </div>
   537→
   538→          <input
   539→            value={title}
   540→            onChange={(e) => onTitleChange(e.target.value)}
   541→            onKeyDown={(e) => {
   542→              if (e.key === "Enter") {
   543→                e.preventDefault();
   544→                editor?.commands.focus("start");
   545→              }
   546→            }}
   547→            placeholder="Untitled"
   548→            aria-label="Document title"
   549→            className="min-w-0 flex-1 rounded-md bg-transparent px-1.5 py-1 text-[14.5px] font-semibold tracking-tight outline-none transition-colors placeholder:text-muted-foreground/50 hover:bg-muted/40 focus:bg-muted/60"
   550→          />
   551→
   552→          <SaveChip status={status} />
   553→
   554→          <div className="flex items-center gap-0.5">
   555→            <HeaderBtn title="Undo — ⌘Z" onClick={onUndo} disabled={!history?.canUndo}>
   556→              <Undo2 size={15} />
   557→            </HeaderBtn>
   558→            <HeaderBtn title="Redo — ⌘⇧Z" onClick={onRedo} disabled={!history?.canRedo}>
   559→              <Redo2 size={15} />
   560→            </HeaderBtn>
   561→
   562→            <span className="mx-1 h-5 w-px bg-border/70" aria-hidden />
   563→
   564→            <HeaderBtn title="Insert demo document" onClick={onLoadDemo}>
   565→              <Presentation size={15} />
   566→            </HeaderBtn>

   643→function StatusBar({
   644→  editor,
   645→  status,
   646→  savedAt,
   647→}: {
   648→  editor: Editor | null;
   649→  status: SaveStatus;
   650→  savedAt: Date | null;
   651→}) {
   652→  const stats = useEditorState({
   653→    editor,
   654→    selector: ({ editor: e }) =>
   655→      e
   656→        ? {
   657→            words: e.storage.characterCount.words(),
   658→            chars: e.storage.characterCount.characters(),
   659→            blocks: e.state.doc.childCount,
   660→          }
   661→        : { words: 0, chars: 0, blocks: 0 },
   662→  });
   663→
   664→  const timeLabel = savedAt
   665→    ? `· ${relativeTime(savedAt)}`
   666→    : "";
   667→
   668→  return (
   669→    <footer className="mt-auto shrink-0 border-t border-border/60 bg-background/85 backdrop-blur-xl">
   670→      <div className="mx-auto flex h-9 w-full max-w-5xl items-center justify-between px-4 pb-[env(safe-area-inset-bottom)] text-[11.5px] text-muted-foreground">
   671→        <div className="flex items-center gap-3 tabular-nums">
   672→          <span>
   673→            <span className="font-medium text-foreground/70">{stats?.blocks ?? 0}</span> blocks
   674→          </span>
   675→          <span aria-hidden>·</span>
   676→          <span>
   677→            <span className="font-medium text-foreground/70">{stats?.words ?? 0}</span> words
   678→          </span>
   679→          <span aria-hidden className="hidden sm:inline">·</span>
   680→          <span className="hidden sm:inline">
   681→            <span className="font-medium text-foreground/70">{stats?.chars ?? 0}</span> characters
   682→          </span>
   683→        </div>
   684→        <div className="flex items-center gap-2">
   685→          <span className="hidden md:inline">Press / for commands · ⌘/ shortcuts</span>
   686→          <span aria-hidden className="hidden md:inline">·</span>
   687→          <span className="flex items-center gap-1.5" role="status" aria-live="polite">
   688→            <span
   689→              className={cn(
   690→                "h-1.5 w-1.5 rounded-full",
   691→                status === "saved" && "bg-emerald-500",
   692→                status === "saving" && "animate-pulse bg-violet-400",
   693→                status === "dirty" && "bg-amber-500",
   694→                status === "error" && "bg-red-400",
   695→                (status === "idle") && "bg-muted-foreground/50",
   696→              )}
   697→            />
   698→            {status === "saved" ? `Saved ${timeLabel}` : status === "idle" ? "All changes captured" : status.charAt(0).toUpperCase() + status.slice(1)}
   699→          </span>
   700→        </div>
   701→      </div>
   702→    </footer>
   703→  );
   704→}

   706→function relativeTime(date: Date): string {
   707→  const seconds = Math.round((Date.now() - date.getTime()) / 1000);
   708→  if (seconds < 10) return "just now";
   709→  if (seconds < 60) return `${seconds}s ago`;
   710→  const minutes = Math.round(seconds / 60);
   711→  if (minutes < 60) return `${minutes}m ago`;
   712→  const hours = Math.round(minutes / 60);
   713→  if (hours < 24) return `${hours}h ago`;
   714→  return date.toLocaleDateString();
   715→}
